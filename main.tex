\begin{frame}{新規スライド}
    frame: 新しいスライドを作成
\end{frame}

\begin{frame}{箇条書き}
    \begin{itemize}
        \item itemize: 箇条書き
        \begin{itemize}
            \item itemizie 内に itemize を入れることも可能
            \footnotemark
        \end{itemize}
    \end{itemize}

    \begin{enumerate}
        \item enumerate: 連番で表示
        \item オプションをつけると好きなように変えられる
        \begin{enumerate}
            \item [A] [A]を指定
            \item [(1)] [(1)]を指定
        \end{enumerate}
    \end{enumerate}

    \footnotetext{
        footnotetext: スライド下部に注釈
    }
\end{frame}

\begin{frame}{数式}
    1. equation: 単一の行で式番号あり
    \begin{equation}
        f(x) = a x^2 + b x + c
    \end{equation}

    2. align: 複数行で、行ごとに式番号あり
    \begin{align}
        x + y &= 5 \\
        2x - y &= 1
    \end{align}

    3. align*: 複数行で、式番号なし（Beamerで最頻出）
    \begin{align*}
        (x+1)^2 &= x^2 + 2x + 1 \\
                &= (x^2 + 1) + 2x
    \end{align*}

    4. equation + aligned: 複数行を綺麗に揃えつつ、全体で「1つの式番号」にする
    \begin{equation}
        \begin{aligned}
            A &= B + C \\
              &= D
        \end{aligned}
    \end{equation}
\end{frame}
